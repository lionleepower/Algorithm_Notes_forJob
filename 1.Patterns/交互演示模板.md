# 交互演示：怎么写一个新的

本库的「动手推演」是放在笔记里的 ` ```html-preview ` 代码块，由 Obsidian 的 **HTML-Preview** 插件渲染成可点击的页面：设置 → 第三方插件 → 搜索 HTML-Preview → 安装并启用，然后在**阅读视图**里点「下一步」。

## 套路：只改 `algo()`

每个演示由两部分组成：

1. **`CODE` 与 `algo()`**：`CODE` 是要高亮的代码行；`algo()` 把算法完整跑一遍，每一步用 `snap(行号, 说明)` 存一份快照。
2. **`render()`**：根据当前快照 `steps[k]` 画出界面（格子、树、数组、变量面板），最后调用 `codeHtml(s)` 和 `nav()`。

其余的按钮、进度条、语法着色、深浅色适配都是通用的，复制过去不用动。写完先在浏览器或 node 里跑一遍，核对最后一步的结果，再放进笔记。

// ===== 只改这里：产出每一步的快照 =====
const CODE=["pre = None","cur = head","while cur:","    nxt = cur.next","    cur.next = pre","    pre = cur","    cur = nxt","return pre"];
const vals=[1,2,3,4];
function algo(){
  const n=vals.length,next=vals.map((_,i)=>i+1<n?i+1:-1);
  let pre=-1,cur=0,nxt=null,steps=[];
  const snap=(line,msg)=>steps.push({line,msg,next:[...next],pre,cur,nxt});
  snap(0,"pre = None：已反转段为空");
  snap(1,"cur = head：指向未反转段的头");

    snap(2,"cur 非空，进入循环");
    nxt=next[cur];snap(3,"先存 nxt，防止下一步断链");
    next[cur]=pre;snap(4,"cur.next 反指向 pre");
    pre=cur;snap(5,"pre 前进到 cur");
    cur=nxt;nxt=null;snap(6,"cur 前进到 nxt");
  }
  snap(2,"cur 为 None，退出循环");
  snap(7,"返回 pre，它才是新头");
  return steps;
}
// ===== 通用部分（不用动）=====
const $=id=>document.getElementById(id);
function hi(line){
  const esc=line.replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;");
  return esc.replace(/(#.*$)|("[^"]*"|'[^']*')|\b(def|class|while|for|in|if|elif|else|return|not|and|or|is|break|continue|import|from|pass)\b|\b(None|True|False|self)\b|\b(\d+)\b/g,

}

}
